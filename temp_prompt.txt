import yfinance as yf

    def collect_training_data(self, min_candles=5000):
        """Collect 5000+ candles for AI training"""
        all_data = {}
        
        for symbol in self.symbols:
            print(f"Fetching {symbol}...")
            
            # 5min candles for 1 year = ~18,000 candles
            df = yf.download(

                auto_adjust=True
            )
            
            if len(df) >= min_candles:
                all_data[symbol] = df
                print(f"✅ {symbol}: {len(df)} candles collected")
            else:
                print(f"❌ {symbol}: Only {len(df)} candles")
        
        return all_data
    
    def prepare_features(self, df):
        """Prepare features for AI training"""
        df = df.copy()
        
        # Price features
        df['returns'] = df['Close'].pct_change()
        df['log_returns'] = np.log(df['Close']/df['Close'].shift(1))
        
        # Volume features
        df['volume_ma'] = df['Volume'].rolling(20).mean()
        df['volume_ratio'] = df['Volume'] / df['volume_ma']
        
        # Volatility
        df['volatility'] = df['returns'].rolling(20).std()
        
        # OHLCV ratios
        df['hl_ratio'] = (df['High'] - df['Low']) / df['Close']
        df['co_ratio'] = (df['Close'] - df['Open']) / df['Open']
        
        # Moving averages
        for period in [5, 10, 20, 50, 200]:
            df[f'sma_{period}'] = df['Close'].rolling(period).mean()
            df[f'ema_{period}'] = df['Close'].ewm(span=period).mean()
        
        # RSI
        df['rsi'] = self.calculate_rsi(df['Close'])
        
        # MACD
        df['macd'], df['signal'] = self.calculate_macd(df['Close'])
        
        # Bollinger Bands
        df['bb_upper'], df['bb_lower'] = self.calculate_bollinger(df['Close'])
        
        df.dropna(inplace=True)
        return df
    
    def calculate_rsi(self, prices, period=14):
        delta = prices.diff()
        gain = (delta.where(delta > 0, 0)).rolling(period).mean()
        loss = (-delta.where(delta < 0, 0)).rolling(period).mean()
        rs = gain / loss
        return 100 - (100 / (1 + rs))
    
    def calculate_macd(self, prices):
        ema12 = prices.ewm(span=12).mean()
        ema26 = prices.ewm(span=26).mean()
        macd = ema12 - ema26
        signal = macd.ewm(span=9).mean()
        return macd, signal
    
    def calculate_bollinger(self, prices, period=20):
        sma = prices.rolling(period).mean()
        std = prices.rolling(period).std()
        return sma + (2 * std), sma - (2 * std)
